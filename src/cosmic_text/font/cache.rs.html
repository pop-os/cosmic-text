3//! Persistent on-disk cache for the parsed system-font index.
4//!
5//! [`fontdb::Database::load_system_fonts`] walks every system font directory and
6//! `mmap`s + parses each font file to populate per-face metadata. The result lives only
7//! in memory and is rebuilt from scratch on every process start.
8//!
9//! This module persists the small, parsed [`fontdb::FaceInfo`] set to disk and reloads it
10//! via [`fontdb::Database::push_face_info`] instead of re-parsing every font file. It is a
11//! pure-Rust, dependency-free reimplementation of what fontconfig's `fc-cache` does.
12//!
13//! The cache is keyed on a cheap fingerprint of the font directories and files that were
14//! scanned (per-directory and per-file modification time, plus file size). On any mismatch
15//! the normal scan runs and the cache is rewritten atomically.
16

25/// Magic bytes identifying a cosmic-text font cache file (`C`osmic `T`ext `F`ont `C`ache).
26const MAGIC: [u8; 4] = *b"CTFC";
27
28/// Cache format version. Bump whenever the on-disk layout changes so that stale caches
29/// written by an older version are rejected instead of misparsed.
30const VERSION: u32 = 1;
31
32/// Loads the system fonts into `db`, using the on-disk cache at `cache_path` when it is
33/// present and still valid.
34///
35/// On a cache hit no font file is parsed: the stored [`FaceInfo`] entries are inserted
36/// directly with [`Database::push_face_info`]. On a miss (or an invalidated cache) the
37/// normal [`Database::load_system_fonts`] scan runs and its result is written back to the
38/// cache for the next start.
39pub(crate) fn load_system_fonts_cached(db: &mut Database, cache_path: &Path) {
40    match try_load(cache_path) {
41        Some(faces) => {
42            log::debug!(
43                "Loaded {} font faces from cache '{}'.",
44                faces.len(),
45                cache_path.display()
46            );
47            for face in faces {
48                db.push_face_info(face);
49            }
50        }
51        None => {
52            db.load_system_fonts();
53            if let Err(err) = write(db, cache_path) {
54                log::warn!(
55                    "Failed to write font cache '{}': {err}",
56                    cache_path.display()
57                );
58            }
59        }
60    }
61}

63/// Returns the default cache file path (`<cache-dir>/cosmic-text/fonts.cache`), following
64/// the platform's conventional cache directory. Returns `None` if no cache directory can
65/// be determined from the environment.
66pub(crate) fn default_cache_path() -> Option<PathBuf> {
67    let mut dir = cache_dir()?;
68    dir.push("cosmic-text");
69    dir.push("fonts.cache");
70    Some(dir)
71}

73#[cfg(all(unix, not(target_os = "macos")))]
74fn cache_dir() -> Option<PathBuf> {
75    if let Some(xdg) = std::env::var_os("XDG_CACHE_HOME") {
76        let path = PathBuf::from(xdg);
77        if path.is_absolute() {
78            return Some(path);
79        }
80    }
81    let mut home = PathBuf::from(std::env::var_os("HOME")?);
82    home.push(".cache");
83    Some(home)
84}

86#[cfg(target_os = "macos")]

88    let mut home = PathBuf::from(std::env::var_os("HOME")?);
89    home.push("Library");
90    home.push("Caches");
91    Some(home)

106/// Reads `cache_path` and, if it is well-formed and its fingerprint still matches the
107/// filesystem, returns the cached set of faces. Returns `None` on any error, version
108/// mismatch, or fingerprint mismatch.
109fn try_load(cache_path: &Path) -> Option<Vec<FaceInfo>> {
110    let bytes = std::fs::read(cache_path).ok()?;
111    let mut r = Reader::new(&bytes);
112
113    if r.take(4)? != MAGIC {
114        return None;
115    }
116    if r.u32()? != VERSION {
117        return None;
118    }
119
120    // Directory fingerprint
121    let dir_count = r.u32()?;
122    for _ in 0..dir_count {
123        let path = r.path()?;
124        let mtime = r.mtime()?;
125        if dir_mtime(&path) != Some(mtime) {
126            return None;
127        }
128    }
129
130    // File fingerprint
131    let file_count = r.u32()? as usize;
132    let mut files = Vec::with_capacity(file_count);
133    for _ in 0..file_count {
134        let path = r.path()?;
135        let mtime = r.mtime()?;
136        let len = r.u64()?;
137        match file_fingerprint(&path) {
138            Some((m, l)) if m == mtime && l == len => {}
139            _ => return None,
140        }
141        files.push(path);
142    }
143
144    // Face table. Each face references a file by index into `files`.
145    let face_count = r.u32()? as usize;
146    let mut faces = Vec::with_capacity(face_count);
147    for _ in 0..face_count {
148        let file_index = r.u32()? as usize;
149        let source_path = files.get(file_index)?.clone();
150        let index = r.u32()?;
151
152        let family_count = r.u32()? as usize;
153        let mut families = Vec::with_capacity(family_count);
154        for _ in 0..family_count {
155            // TODO: Language is not persisted, but it is also not used
156            families.push((r.string()?, fontdb::Language::Unknown));
157        }
158
159        let post_script_name = r.string()?;
160        let style = decode_style(r.u8()?)?;
161        let weight = Weight(r.u16()?);
162        let stretch = decode_stretch(r.u8()?)?;
163        let monospaced = r.u8()? != 0;
164
165        faces.push(FaceInfo {
166            id: ID::dummy(),
167            source: Source::File(source_path),
168            index,
169            families,
170            post_script_name,
171            style,
172            weight,
173            stretch,
174            monospaced,
175        });
176    }
177
178    Some(faces)
179}

181/// Serializes the file-backed faces in `db` to `cache_path`, writing atomically
182fn write(db: &Database, cache_path: &Path) -> std::io::Result<()> {
183    let mut file_indices: HashMap<PathBuf, u32> = HashMap::default();
184    let mut files: Vec<PathBuf> = Vec::new();
185    let mut faces: Vec<(u32, &FaceInfo)> = Vec::new();
186
187    for face in db.faces() {
188        // cosmic-text's `std` feature always enables `fontdb/memmap`
189        let path = match &face.source {
190            Source::File(path) => path,
191            Source::SharedFile(path, _) => path,
192            // Binary sources (e.g. user-provided fonts) are not persistable.
193            Source::Binary(_) => continue,
194        };
195        let index = *file_indices.entry(path.clone()).or_insert_with(|| {
196            let i = files.len() as u32;
197            files.push(path.clone());
198            i
199        });
200        faces.push((index, face));
201    }
202
203    // Directories that directly contained a scanned font, deduplicated.
204    let mut dirs: Vec<PathBuf> = Vec::new();
205    let mut seen_dirs: HashSet<PathBuf> = HashSet::default();
206    for path in &files {
207        if let Some(parent) = path.parent() {
208            if seen_dirs.insert(parent.to_path_buf()) {
209                dirs.push(parent.to_path_buf());
210            }
211        }
212    }
213
214    let mut w = Writer::new();
215    w.bytes(&MAGIC);
216    w.u32(VERSION);
217
218    w.u32(dirs.len() as u32);
219    for dir in &dirs {
220        w.path(dir);
221        w.mtime(dir_mtime(dir).unwrap_or((0, 0)));
222    }
223
224    w.u32(files.len() as u32);
225    for path in &files {
226        w.path(path);
227        let (mtime, len) = file_fingerprint(path).unwrap_or(((0, 0), 0));
228        w.mtime(mtime);
229        w.u64(len);
230    }
231
232    w.u32(faces.len() as u32);
233    for (file_index, face) in &faces {
234        w.u32(*file_index);
235        w.u32(face.index);
236        w.u32(face.families.len() as u32);
237        for (name, _language) in &face.families {
238            w.string(name);
239        }
240        w.string(&face.post_script_name);
241        w.u8(encode_style(face.style));
242        w.u16(face.weight.0);
243        w.u8(face.stretch.to_number() as u8);
244        w.u8(u8::from(face.monospaced));
245    }
246
247    write_atomic(cache_path, &w.into_inner())
248}

250/// Writes `data` to `path` atomically
251fn write_atomic(path: &Path, data: &[u8]) -> std::io::Result<()> {
252    if let Some(parent) = path.parent() {
253        std::fs::create_dir_all(parent)?;
254    }
255
256    let mut tmp = path.as_os_str().to_os_string();
257    // A per-process suffix keeps concurrent writers from writing to eachother's temp file
258    tmp.push(format!(".{}.tmp", std::process::id()));
259    let tmp = PathBuf::from(tmp);
260
261    std::fs::write(&tmp, data)?;
262    match std::fs::rename(&tmp, path) {
263        Ok(()) => Ok(()),
264        Err(err) => {
265            let _ = std::fs::remove_file(&tmp);
266            Err(err)
267        }
268    }
269}

271/// Returns the modification time of a directory as `(secs, nanos)` since the Unix epoch.
272fn dir_mtime(path: &Path) -> Option<(u64, u32)> {
273    let metadata = std::fs::metadata(path).ok()?;
274    if !metadata.is_dir() {
275        return None;
276    }
277    mtime_parts(metadata.modified().ok()?)
278}
279
280/// Returns `((secs, nanos), len)` for a regular file, or `None` if it is missing or not a file.
281fn file_fingerprint(path: &Path) -> Option<((u64, u32), u64)> {
282    let metadata = std::fs::metadata(path).ok()?;
283    if !metadata.is_file() {
284        return None;
285    }
286    Some((mtime_parts(metadata.modified().ok()?)?, metadata.len()))
287}
288
289fn mtime_parts(time: SystemTime) -> Option<(u64, u32)> {
290    let dur = time.duration_since(SystemTime::UNIX_EPOCH).ok()?;
291    Some((dur.as_secs(), dur.subsec_nanos()))
292}

294fn encode_style(style: Style) -> u8 {
295    match style {
296        Style::Normal => 0,
297        Style::Italic => 1,
298        Style::Oblique => 2,
299    }
300}
301
302fn decode_style(value: u8) -> Option<Style> {
303    match value {
304        0 => Some(Style::Normal),
305        1 => Some(Style::Italic),
306        2 => Some(Style::Oblique),
307        _ => None,
308    }
309}
310
311fn decode_stretch(value: u8) -> Option<Stretch> {
312    Some(match value {
313        1 => Stretch::UltraCondensed,
314        2 => Stretch::ExtraCondensed,
315        3 => Stretch::Condensed,
316        4 => Stretch::SemiCondensed,
317        5 => Stretch::Normal,
318        6 => Stretch::SemiExpanded,
319        7 => Stretch::Expanded,
320        8 => Stretch::ExtraExpanded,
321        9 => Stretch::UltraExpanded,
322        _ => return None,
323    })
324}

326/// Append-only little-endian byte writer.
327struct Writer {
328    buf: Vec<u8>,
329}
330
331impl Writer {
332    fn new() -> Self {
333        Self { buf: Vec::new() }
334    }
335
336    fn into_inner(self) -> Vec<u8> {
337        self.buf
338    }
339
340    fn bytes(&mut self, value: &[u8]) {
341        self.buf.extend_from_slice(value);
342    }
343
344    fn u8(&mut self, value: u8) {
345        self.buf.push(value);
346    }
347
348    fn u16(&mut self, value: u16) {
349        self.buf.extend_from_slice(&value.to_le_bytes());
350    }
351
352    fn u32(&mut self, value: u32) {
353        self.buf.extend_from_slice(&value.to_le_bytes());
354    }
355
356    fn u64(&mut self, value: u64) {
357        self.buf.extend_from_slice(&value.to_le_bytes());
358    }
359
360    fn mtime(&mut self, (secs, nanos): (u64, u32)) {
361        self.u64(secs);
362        self.u32(nanos);
363    }
364
365    fn string(&mut self, value: &str) {
366        self.u32(value.len() as u32);
367        self.bytes(value.as_bytes());
368    }
369
370    /// Encodes a path as its lossy UTF-8 string form
371    fn path(&mut self, path: &Path) {
372        self.string(&path.to_string_lossy());
373    }
374}

376/// Bounds-checked little-endian byte reader. Every accessor returns `None` on truncation
377/// or malformed data so a corrupt cache is rejected rather than panicking.
378struct Reader<'a> {
379    data: &'a [u8],
380    pos: usize,
381}
382
383impl<'a> Reader<'a> {
384    fn new(data: &'a [u8]) -> Self {
385        Self { data, pos: 0 }
386    }
387
388    fn take(&mut self, len: usize) -> Option<&'a [u8]> {
389        let end = self.pos.checked_add(len)?;
390        let slice = self.data.get(self.pos..end)?;
391        self.pos = end;
392        Some(slice)
393    }
394
395    fn u8(&mut self) -> Option<u8> {
396        Some(self.take(1)?[0])
397    }
398
399    fn u16(&mut self) -> Option<u16> {
400        Some(u16::from_le_bytes(self.take(2)?.try_into().ok()?))
401    }
402
403    fn u32(&mut self) -> Option<u32> {
404        Some(u32::from_le_bytes(self.take(4)?.try_into().ok()?))
405    }
406
407    fn u64(&mut self) -> Option<u64> {
408        Some(u64::from_le_bytes(self.take(8)?.try_into().ok()?))
409    }
410
411    fn mtime(&mut self) -> Option<(u64, u32)> {
412        Some((self.u64()?, self.u32()?))
413    }
414
415    fn string(&mut self) -> Option<String> {
416        let len = self.u32()? as usize;
417        let bytes = self.take(len)?;
418        String::from_utf8(bytes.to_vec()).ok()
419    }
420
421    fn path(&mut self) -> Option<PathBuf> {
422        Some(PathBuf::from(self.string()?))
423    }
